    void LateUpdate()
    {
        if (!rec || cam == null) return;

"left" : mn == r0 ? "right" : mn == t0 ? "top" : "bottom"; } } bool ok = In(p); if (ok && !HudAll.Contains(p)) visAll++; if (ok && !HudBars.Contains(p)) visBars++; }
        a.minRingVisAll = Mathf.Min(a.minRingVisAll, visAll / 1.8f); a.minRingVisBars = Mathf.Min(a.minRingVisBars, visBars / 1.8f); a.minMargin = Mathf.Min(a.minMargin, minEdge);

        {

    }

    IEnumerator Seq()
    {
        enemy.enabled = true; enc.Restart(); pH.Revive(); Keys(); Place(player, Vector3.zero); iso.Snap(); rec = true;
        phase = "idle"; yield return Wait(0.5f); yield return Shot("1_idle");
        // run toward enemy (enemy approaches from spawn)
        phase = "run_toward_enemy"; float t0 = Time.time; bool shotRun = false;
        while (Time.time - t0 < 2.5f) { Vector3 to = enemyT.position - player.position; to.y = 0; if (to.magnitude < 3.2f) break; Keys(Steer(to.normalized)); if (!shotRun && Time.time - t0 > 0.5f) { shotRun = true; Keys(Steer(to.normalized)); yield return Shot("2_run"); } yield return null; }
        // dash sideways (screen right)
        phase = "dash_sideways"; Keys(new[] { Key.D, Key.Space }); yield return null; yield return null; Keys(Key.D); yield return Wait(0.05f); yield return Shot("3_dash"); yield return Wait(0.3f); Keys();
        // attack
        phase = "attack"; t0 = Time.time; bool shotAtk = false;
        while (Time.time - t0 < 1.4f) { Vector3 to = enemyT.position - player.position; to.y = 0; Keys(Steer(to.magnitude > 1.7f ? to.normalized : Vector3.zero, true)); if (!shotAtk && Time.time - t0 > 0.55f) { shotAtk = true; yield return Shot("4_attack"); } yield return null; }
        Keys();
        // wait for enemy windup then dodge through it
        phase = "wait_for_windup"; t0 = Time.time; bool d = false;
        while (Time.time - t0 < 6f)
        {
            if (enemy.IsTelegraphing && enemy.WindupProgress > 0.45f && !d)
            {
                d = true; phase = "dodge_dash"; yield return Shot("5_telegraph");
                Vector3 to = enemyT.position - player.position; to.y = 0; var perp = Vector3.Cross(Vector3.up, to.normalized);
                Keys(Steer(perp, false, true)); yield return null; yield return null; Keys(Steer(perp)); yield return Wait(0.06f); yield return Shot("6_dodge"); yield return Wait(0.3f); break;
            }
            Vector3 q = enemyT.position - player.position; q.y = 0; Keys(Steer(q.magnitude > 2.4f ? q.normalized : Vector3.zero)); yield return null;
        }
        Keys(); rec = false;
    }

    IEnumerator MeasureRoutine(string label, float ramp)
    {

        Hold(ramp); for (int i = 0; i < 4; i++) yield return null;
        yield return Cap(outDir + "/" + lightDir + "/" + label + ".png"); string capInfo = capSize.Replace(",", ";");
        Hold(ramp); yield return null; yield return null; yield return null; preview.enabled = false;
        var tr = TeleRoot(); tr.SetActive(false);
        var A = Grab().GetPixels32();                         // actors on, telegraph off
        foreach (var r in eR) r.enabled = false; var P = Grab().GetPixels32();   // player only
        foreach (var r in pR) r.enabled = false; var F = Grab().GetPixels32();   // floor only
        tr.SetActive(true); var FT = Grab().GetPixels32();     // floor + telegraph
        foreach (var r in pR) r.enabled = true; foreach (var r in eR) r.enabled = true; preview.enabled = true;
        // floor mask: ground points within 8 m of origin (combat area), stride 2
        double fs = 0; int fn = 0; var floorMask = new bool[W * H]; var lifts = new List<float>();
        for (int y = 0; y < H; y += 2) for (int x = 0; x < W; x += 2)
        {
            var ray = cam.ScreenPointToRay(new Vector3(x, H - 1 - y, 0)); if (Mathf.Abs(ray.direction.y) < 1e-4f) continue; float t = -ray.origin.y / ray.direction.y; if (t < 0) continue; var gp = ray.origin + ray.direction * t;
            if (new Vector2(gp.x - holdCenter.x, gp.z - holdCenter.z).magnitude < 8f) { int i = (H - 1 - y) * W + x; floorMask[i] = true; }
        }
        // note: ScreenPointToRay y is bottom-up; pixel array from ReadPixels is bottom-up as well, so index (H-1-y)*W+x is consistent
        var silP = new bool[W * H]; var silE = new bool[W * H];
        for (int i = 0; i < W * H; i++)
        {
            if (floorMask[i]) { bool chr = Mathf.Abs(Lum(A[i]) - Lum(F[i])) > 0.012f; if (!chr) { fs += Lum(F[i]); fn++; } }
            if (Mathf.Abs(Lum(P[i]) - Lum(F[i])) > 0.012f) silP[i] = true;
            if (Mathf.Abs(Lum(A[i]) - Lum(P[i])) > 0.012f) silE[i] = true;
            if (floorMask[i]) { float d = Lum(FT[i]) - Lum(F[i]); if (d > 0.004f) lifts.Add(d); }
        }
        float floorMean = (float)(fs / System.Math.Max(1, fn));
        System.Func<bool[], float[]> sep = (sil) =>
        {
